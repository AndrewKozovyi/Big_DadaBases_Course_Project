// MDX-запити для демонстрації куба «Оренда» (база Rental_SSAS).
// У SSMS: підключення Analysis Services -> Rental_SSAS -> New Query -> MDX; виділити запит і натиснути F5.

// 1. Загальні підсумки (звірка з DataWarehouse/ETL_REPORTS.sql, розділ 6)
SELECT {[Measures].[Сума рахунків], [Measures].[Сплачено], [Measures].[Борг], [Measures].[Кількість рахунків],
        [Measures].[Сума платежів], [Measures].[Кількість платежів], [Measures].[Дохід від оренди],
        [Measures].[Кількість договорів], [Measures].[Унікальних орендарів], [Measures].[Орендованих приміщень]} ON COLUMNS
FROM [Оренда];

// 2. Time Intelligence по роках: попередній рік (PY), приріст (YoY), рівень оплати, завантаженість
SELECT {[Measures].[Сума рахунків], [Measures].[Сума рахунків PY], [Measures].[Сума рахунків YoY],
        [Measures].[Рівень оплати], [Measures].[Завантаженість], [Measures].[Частка вчасних платежів]} ON COLUMNS,
       [Період].[Календар].[Рік].Members ON ROWS
FROM [Оренда];

// 3. Помісячна динаміка 2025 року: накопичений підсумок (YTD), MoM та ковзне середнє за 3 місяці
SELECT {[Measures].[Сума рахунків], [Measures].[Сума рахунків YTD], [Measures].[Сума платежів],
        [Measures].[Сума платежів MoM], [Measures].[Платежі ковзне середнє 3], [Measures].[Завантаженість ковзна 3]} ON COLUMNS,
       Descendants([Період].[Календар].[Рік].&[2025], [Період].[Календар].[Місяць]) ON ROWS
FROM [Оренда];

// 4. Ranking: топ-10 боржників і їхнє місце в рейтингу
SELECT {[Measures].[Борг], [Measures].[Місце орендаря за боргом], [Measures].[Кількість рахунків],
        [Measures].[Частка прострочених рахунків]} ON COLUMNS,
       [Топ-10 боржників] ON ROWS
FROM [Оренда];

// 5. Ranking: топ-10 приміщень за доходом, місце та частка в загальному доході
SELECT {[Measures].[Дохід від оренди], [Measures].[Місце приміщення за доходом],
        [Measures].[Частка доходу від загального], [Measures].[Завантаженість]} ON COLUMNS,
       [Топ-10 приміщень за доходом] ON ROWS
FROM [Оренда];

// 6. KPI: значення, ціль, статус (1 / 0 / -1) і тренд порівняно з попереднім роком
SELECT {KPIValue("Рівень оплати рахунків"), KPIGoal("Рівень оплати рахунків"),
        KPIStatus("Рівень оплати рахунків"), KPITrend("Рівень оплати рахунків"),
        KPIValue("Завантаженість приміщень"), KPIGoal("Завантаженість приміщень"),
        KPIStatus("Завантаженість приміщень"), KPITrend("Завантаженість приміщень"),
        KPIValue("Своєчасність оплат"), KPIStatus("Своєчасність оплат")} ON COLUMNS,
       [Період].[Календар].[Рік].Members ON ROWS
FROM [Оренда];

// 7. Обчислюваний член виміру «Комерційна нерухомість» поряд із групами типів
SELECT {[Measures].[Дохід від оренди], [Measures].[Завантаженість], [Measures].[Середній рахунок]} ON COLUMNS,
       {[Приміщення].[Група типів].[Група типів].Members,
        [Приміщення].[Група типів].[Комерційна нерухомість]} ON ROWS
FROM [Оренда];

// 8. SCD2: історичні та поточні версії приміщень (після 07_INCREMENTAL_TEST_DATA.sql)
SELECT {[Measures].[Сума рахунків], [Measures].[Кількість рахунків]} ON COLUMNS,
       NON EMPTY ([Приміщення].[Версія запису].[Версія запису].Members,
                  [Приміщення].[Тип приміщення].[Тип приміщення].Members) ON ROWS
FROM [Оренда];

// 9. Ієрархія Місто -> Приміщення: 5 найприбутковіших міст
SELECT {[Measures].[Дохід від оренди], [Measures].[Орендовані дні], [Measures].[Завантаженість]} ON COLUMNS,
       TopCount([Приміщення].[Географія].[Місто].Members, 5, [Measures].[Дохід від оренди]) ON ROWS
FROM [Оренда];

// 10. Перспектива «Фінанси»: своєчасність оплати за типом орендаря
SELECT {[Measures].[Сума платежів], [Measures].[Кількість платежів], [Measures].[Середнє запізнення оплати днів]} ON COLUMNS,
       ([Орендар].[Тип орендаря].[Тип орендаря].Members,
        [Своєчасність оплати].[Своєчасність оплати].[Своєчасність оплати].Members) ON ROWS
FROM [Фінанси];
