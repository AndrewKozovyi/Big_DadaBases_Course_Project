// Запит 1. Сума рахунків і сплачено по роках і групах типів приміщень
SELECT {[Measures].[Сума рахунків], [Measures].[Сплачено]} ON COLUMNS,
       NON EMPTY ([Період].[Календар].[Рік].Members, [Приміщення].[Група типів].[Група типів].Members) ON ROWS
FROM [Оренда];

// Запит 2. Топ-10 боржників
SELECT {[Measures].[Борг]} ON COLUMNS,
       TopCount([Орендар].[Орендар].[Орендар].Members, 10, [Measures].[Борг]) ON ROWS
FROM [Оренда];

// Запит 3. Завантаженість і дохід по містах і місяцях 2025 року
SELECT {[Measures].[Завантаженість], [Measures].[Дохід від оренди]} ON COLUMNS,
       NON EMPTY ([Приміщення].[Місто].[Місто].Members,
                  Descendants([Період].[Календар].[Рік].&[2025], [Період].[Календар].[Місяць])) ON ROWS
FROM [Оренда];
